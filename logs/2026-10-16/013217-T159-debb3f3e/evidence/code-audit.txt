10: export function checkThumbnails(dir: string): ThumbResult[] {
11:   const path = join(dir, 'thumbnails-report.json');
12:   if (!existsSync(path)) throw Error('thumbnails-report.jsonがない');
13:   const report: {file: string; variant: string; headlineSize: number; fitted: boolean; contrast: number}[] = JSON.parse(readFileSync(path, 'utf8'));
14:   return report.map(r => {
15:     const issues: string[] = [];
16:     const bytes = existsSync(join(dir, r.file)) ? statSync(join(dir, r.file)).size : 0;
17:     if (!bytes) issues.push('画像がない');
18:     if (bytes > THUMB_MAX_BYTES) issues.push(`${THUMB_MAX_BYTES}バイト（モバイル上限）を超える`);
19:     if (!r.fitted) issues.push('見出しが収まらない');
20:     if (r.contrast < MIN_CONTRAST) issues.push(`コントラスト比${r.contrast.toFixed(2)}が${MIN_CONTRAST}未満`);
21:     return {file: r.file, variant: r.variant, bytes, contrast: Number(r.contrast.toFixed(2)), headlineSize: r.headlineSize, ok: !issues.length, issues};
22:   });
23: }

158: export function reviseThumbnail(dir: string, to: string, reason: string, by = 'cli-user'): Revision {
159:   const path = join(dir, 'metadata.json');
160:   if (!existsSync(path)) throw Error('metadata.jsonがない');
161:   if (!reason.trim()) throw Error('見直しの理由（--reason）が必要');
162:   if (!/^thumbnail-[a-c]\.png$/.test(to)) throw Error('サムネイルは thumbnail-a.png, thumbnail-b.png, thumbnail-c.png のいずれか');
163:   const m: Metadata = JSON.parse(readFileSync(path, 'utf8'));
164:   const from = m.selectedThumbnail ?? 'thumbnail-a.png';
165:   const rev: Revision = {at: new Date().toISOString(), field: 'thumbnail', from, to, reason, by};
166:   const histPath = join(dir, 'metadata-revisions.json');
167:   const hist: Revision[] = existsSync(histPath) ? JSON.parse(readFileSync(histPath, 'utf8')) : [];
168:   writeFileSync(histPath, JSON.stringify([...hist, rev], null, 2));
169:   writeFileSync(path, JSON.stringify({...m, selectedThumbnail: to}, null, 2));
170:   return rev;
171: }
